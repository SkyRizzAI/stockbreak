// Decode Anchor `emit!` events from transaction logs with Kit + Codama-generated decoders.
// Anchor logs events as: "Program data: <base64(discriminator[8] || borsh(event))>"
// Logs from CPI'd programs interleave, so track the invoke stack and only take lines
// emitted while OUR program is the currently executing program.
import { getBase64Encoder, type Address, type ReadonlyUint8Array } from '@solana/kit';
import {
  DEMO_PROGRAM_ADDRESS,
  DemoEvent,
  identifyDemoEvent,
  parseCounterInitializedEvent,
} from '../clients/demo/src/generated';

const INVOKE = /^Program (\w+) invoke \[(\d+)\]$/;
const EXIT = /^Program (\w+) (success|failed)/;
const DATA = 'Program data: ';

export function extractProgramDataFromLogs(logs: readonly string[], programId: Address): ReadonlyUint8Array[] {
  const stack: string[] = [];
  const out: ReadonlyUint8Array[] = [];
  const b64 = getBase64Encoder();
  for (const line of logs) {
    const inv = INVOKE.exec(line);
    if (inv) { stack.push(inv[1]!); continue; }
    if (EXIT.test(line)) { stack.pop(); continue; }
    if (line.startsWith(DATA) && stack.at(-1) === programId) {
      try { out.push(b64.encode(line.slice(DATA.length))); } catch { /* ignore malformed */ }
    }
  }
  return out;
}

export type DecodedDemoEvent = { kind: 'CounterInitialized'; data: ReturnType<typeof parseCounterInitializedEvent> };

export function decodeDemoEvents(logs: readonly string[]): DecodedDemoEvent[] {
  const events: DecodedDemoEvent[] = [];
  for (const bytes of extractProgramDataFromLogs(logs, DEMO_PROGRAM_ADDRESS)) {
    let kind: DemoEvent;
    try { kind = identifyDemoEvent(bytes); } catch { continue; } // not one of ours
    switch (kind) {
      case DemoEvent.CounterInitialized:
        events.push({ kind: 'CounterInitialized', data: parseCounterInitializedEvent(bytes) });
        break;
    }
  }
  return events;
}
