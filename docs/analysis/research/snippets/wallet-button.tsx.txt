'use client';
import { useClient, useAction } from '@solana/react';
import { useConnect, useConnectedWallet, useDisconnect, useWallets, WalletReadyGate, useSignMessage } from '@solana/kit-plugin-wallet/react';
import type { AppClient } from './providers';
import { getInitializeInstructionAsync } from '../clients/demo/src/generated';

function Inner() {
  const client = useClient<AppClient>();
  const wallets = useWallets(client);
  const connected = useConnectedWallet(client);
  const { dispatch: connect, isRunning } = useConnect(client);
  const { dispatch: disconnect } = useDisconnect(client);
  const { dispatch: sign } = useSignMessage(client);
  const send = useAction(async (signal: AbortSignal) => {
    const ix = await getInitializeInstructionAsync({ authority: client.identity, start: 1n });
    const res = await client.sendTransaction([ix], { abortSignal: signal });
    return res.context.signature;
  });
  if (!connected)
    return (
      <>
        {wallets.map((w) => (
          <button key={w.name} disabled={isRunning} onClick={() => connect(w)}>
            {w.name}
          </button>
        ))}
      </>
    );
  return (
    <div>
      {connected.account.address} v0:{String(connected.supportedTransactionVersions.has(0))}
      <button onClick={() => sign(new TextEncoder().encode('hello'))}>Sign</button>
      <button onClick={() => send.dispatch()}>Init</button>
      <button onClick={() => disconnect()}>Disconnect</button>
    </div>
  );
}
export function WalletButton() {
  const client = useClient<AppClient>();
  return (
    <WalletReadyGate client={client} fallback={<span>Loading wallets…</span>}>
      <Inner />
    </WalletReadyGate>
  );
}
